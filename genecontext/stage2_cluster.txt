function clusterGeneSymbol(group){
 const raw=(group.find(m=>m.source==='annotation'&&m.annotation_name)?.annotation_name||groupLabel(group)).split(' / ')[0].trim(),symbol=raw.replace(/\s+family candidate.*$/i,'').replace(/\s*\([^)]*\).*$/,'').trim();

}
function clusterSourceBadge(group){

function classicStructures(r,group){
 const result=new Map();for(const m of group){
  const features=m.source==='annotation'?(DATA.annotation_features||[]).filter(f=>f.region_id===r.region_id&&f.original_gene===m.original_gene&&f.contig===m.contig&&f.strand===m.strand):[],unique=[...new Map(features.map(f=>[[f.start,f.end].join(':'),f])).values()],feature=unique.length===1?unique[0]:null;
  const v={...m,model_start:m.start,model_end:m.end,start:feature?feature.start:m.start,end:feature?feature.end:m.end,span_origin:feature?'GFF gene boundary':'saved model span'},key=[v.source,v.start,v.end,v.strand,!!v.context_only,!!v.diagnostic_fragment].join(':');if(!result.has(key))result.set(key,[]);result.get(key).push(v);

function classicRuler(r,left,right){
 const layout=neighborhood(r,'all');
 const structures=[...layout.left,...layout.anchor,...layout.right].flatMap(group=>classicStructures(r,group));
 const annotation=structures.filter(group=>group[0].source==='annotation');
 const w=windowRange(r),spans=(annotation.length?annotation:structures).map(group=>groupRange(r,group));
 const lo=Math.max(w[0],Math.min(0,...spans.map(span=>span[0])));
 const hi=Math.min(w[1],Math.max(0,...spans.map(span=>span[1])));
 const pad=Math.max(1,(hi-lo)*.025);
 const window=[Math.max(w[0],lo-pad),Math.min(w[1],hi+pad)];
 return classicLinearRuler(window,left,right);
}
function classicLinearRuler(window,left,right){
 const span=Math.max(1,window[1]-window[0]);
 const pxPerBp=(right-left)/span,ideal=span/Math.max(2,(right-left)/140);
 const power=10**Math.floor(Math.log10(Math.max(1,ideal))),n=ideal/power;
 const major=(n<=1?1:n<=2?2:n<=5?5:10)*power;
 return {window,pxPerBp,major,minor:major/5,center:left-window[0]*pxPerBp,bx:bp=>left+(bp-window[0])*pxPerBp};
}
function classicNearestNeighbors(region){
 const features=(DATA.annotation_features||[]).filter(gene=>gene.region_id===region.region_id&&gene.contig===region.contig&&gene.biotype==='protein_coding');
 const candidates=[null,null];
 for(const gene of features){
  const side=gene.end<=region.anchor_start?0:gene.start>=region.anchor_end?1:-1;
  if(side<0)continue;
  const gap=side===0?region.anchor_start-gene.end:gene.start-region.anchor_end;
  if(!candidates[side]||gap<candidates[side].gap)candidates[side]={gap,name:gene.name||gene.original_gene};
 }
 return {available:features.length>0, sides:region.strand==='-'?candidates.reverse():candidates};
}
