// Focused linear coordinates; source coverage is a gene overview, not a transcript.
function neighborhoodDiagram(rows,edges){
 const view=$('scale').value;
 return view==='distance'?clusterNeighborhoodDiagram(rows,edges):view==='models'?detailedNeighborhoodDiagram(rows,edges):schematicNeighborhoodDiagram(rows,edges);
}
function clusterGeneSymbol(group){
 const raw=groupLabel(group).split(' / ')[0].trim(),symbol=raw.replace(/\s+family candidate.*$/i,'').replace(/\s*\([^)]*\).*$/,'').trim();
 if(/^LOC\d+$/.test(symbol))return 'LOC…'+symbol.slice(-4);
 return /^[A-Za-z][A-Za-z0-9_.-]{0,17}$/.test(symbol)?symbol:'?';
}
function clusterSourceBadge(group){
 return ['annotation','tiberius'].map(source=>{const n=group.filter(m=>m.source===source).length;return n?(source==='annotation'?'A':'T')+(n>1?'×'+n:''):''}).filter(Boolean).join(' / ');
}
function clusterCoverage(group){
 const blocks=group.flatMap(m=>m.cds_blocks||[]).map(b=>[...b]).sort((a,b)=>a[0]-b[0]||a[1]-b[1]),out=[];
 for(const b of blocks){const last=out[out.length-1];if(last&&b[0]<=last[1])last[1]=Math.max(last[1],b[1]);else out.push(b)}
 return out;
}
function clusterRowRuler(r,left,right){
 // The focus is calculated from BOTH sources, even when one source is hidden.
 const groups=['all','annotation','tiberius'].flatMap(source=>{const l=neighborhood(r,source);return [...l.left,...l.anchor,...l.right]}),nativeWindow=windowRange(r);
 const spans=groups.map(g=>groupRange(r,g)).filter(a=>a[1]>=nativeWindow[0]&&a[0]<=nativeWindow[1]);
 let lo=Math.max(nativeWindow[0],Math.min(0,...spans.map(a=>a[0]))),hi=Math.min(nativeWindow[1],Math.max(0,...spans.map(a=>a[1])));
 if(!(hi>lo)){lo=nativeWindow[0];hi=nativeWindow[1]}
 const pad=(hi-lo)*.025,window=[Math.max(nativeWindow[0],lo-pad),Math.min(nativeWindow[1],hi+pad)],span=window[1]-window[0],pxPerBp=(right-left)/span;
 const ideal=span/Math.max(2,(right-left)/140),magnitude=10**Math.floor(Math.log10(ideal)),normal=ideal/magnitude,major=(normal<=1?1:normal<=2?2:normal<=5?5:10)*magnitude;
 return {window,pxPerBp,major,minor:major/5,center:left-window[0]*pxPerBp,bx:bp=>left+(bp-window[0])*pxPerBp};
}
