function validateResult(next,expected){

async function pollHosted(initial){

 try{
  for(;;){

   if(job.status==='failed')throw Error(job.step||'Alignment failed');
   status.textContent='Recalculating… '+(job.step||'MAFFT is running')+'. You may reload the page; the calculation continues.';
   await new Promise(resolve=>setTimeout(resolve,1000));
   const state=await(await api('status')).json();if(!state.job||state.job.job_id!==initial.job_id)throw Error('The selection was replaced in another tab. Reload to show its latest result.');job=state.job;
  }
 }catch(e){status.textContent=e.message+' The previous alignment and decisions were retained. Reload to reconnect to a running job.'}
 finally{polling=false;setRunning(false)}
}
