function codingExons(source){
 const m=source?.metadata,native=source?.cds,translated=(source?.protein?.length||0)*3;
 if(source?.reference||!native||!translated||!Array.isArray(m?.cds_blocks)||!m.cds_blocks.length||!['+','-'].includes(m.strand))return [];
 if(native.length!==translated&&!(native.length===translated+3&&/^(TAA|TAG|TGA)$/.test(native.slice(-3))))return [];
 if(m.cds_blocks.some(b=>!Array.isArray(b)||!Number.isSafeInteger(b[0])||!Number.isSafeInteger(b[1])||b[0]<0||b[1]<=b[0]))return [];
 const ordered=[...m.cds_blocks].sort((a,b)=>a[0]-b[0]);
 if(m.strand==='-')ordered.reverse();
 const trim=ordered[0][2]??0;
 if(![0,1,2].includes(trim)||trim>=ordered[0][1]-ordered[0][0])return [];
 const exons=[];let total=0;
 for(const [start,end] of ordered){
  const previous=exons.at(-1),gap=previous?(m.strand==='-'?previous.start-end:start-previous.end):null;
  if(gap!==null&&gap<0)return [];
  const size=end-start-(previous?0:trim);
  if(gap===0){previous.start=Math.min(previous.start,start);previous.end=Math.max(previous.end,end);previous.to+=size;}
  else exons.push({start,end,from:total,to:total+size,intron:gap});
  total+=size;
 }
 if(total<translated||total>native.length)return [];
 return exons.filter(exon=>exon.from<translated).map(exon=>({...exon,to:Math.min(exon.to,translated)}));
}

function markExonCell(cell,exons,offset,width,cursor){
 while(exons[cursor].to<=offset)cursor++;
 const ticks=[],stops=[],details=[];
 for(let i=cursor;i<exons.length&&exons[i].from<offset+width;i++){
  const exon=exons[i],start=Math.max(0,exon.from-offset),end=Math.min(width,exon.to-offset);
  const left=100*start/width,right=100*end/width;
  stops.push((i%2?'#599b9a':'#226b84')+' '+left+'% '+right+'%');
  details.push('CDS exon '+(i+1));
  if(i&&exon.from>=offset){
   ticks.push(`linear-gradient(90deg,transparent calc(${left}% - 1px),#9e5513 calc(${left}% - 1px),#9e5513 calc(${left}% + 1px),transparent calc(${left}% + 1px))`);
   details.push(`Intron ${i} · ${exon.intron.toLocaleString('en-US')} bp${width===3&&start?' · after base '+start+' of this codon':''}`);
  }
 }
 cell.classList.add('gcf-exon-cell');cell.style.setProperty('--exon-stripe',stops.length===1?(cursor%2?'#599b9a':'#226b84'):`linear-gradient(90deg,${stops.join(',')})`);
 if(ticks.length)cell.style.setProperty('--splice-ticks',ticks.join(','));
 cell.title=details.join(' · ');
 return cursor;
}
