function pendingComparisonChoice(r,m){
 const p=reviewPlan(r),key=p.primary_model===m.model_id?'primary':m.model_id,pending=p.form_drafts?.[key];return pending?.model_id===m.model_id?pending:null;
}
function toggleAlignmentCandidate(r,m,checked){
 const p=reviewPlan(r),pending=pendingComparisonChoice(r,m);if(p.form_drafts)delete p.form_drafts[p.primary_model===m.model_id?'primary':m.model_id];state.review_plan[r.region_id]=p;if(pending)state.models[m.model_id]={decision:'not_selected',role:'',...(state.models[m.model_id]||{}),notes:pending.notes,copy_label:pending.copy_label||'',comparison_group:pending.comparison_group};
 if(checked){
  const old=state.models[m.model_id]||{};
  if(old.decision!=='inspect')state.models[m.model_id]={...old,decision:'inspect',role:'alignment_candidate',notes:pending?.notes??old.notes??'',target_override:false,comparison_group:r.gene||'target',copy_label:old.copy_label||''};
  const p=reviewPlan(r);if(m.model_id!==p.primary_model&&!p.additional.includes(m.model_id))p.additional.push(m.model_id);
  p.additional_roles={...p.additional_roles,[m.model_id]:'alignment_candidate'};state.review_plan[r.region_id]=p;
 }else clearModelChoice(m);
 const region=state.regions[r.region_id];if(region)region.saved=false;
 persisted();renderReviewQueue();
}
function cdsComparisonScale(values,mode='exons',zoom=1,availableWidth=800){
 const lo=Math.min(...values.map(m=>m.start)),hi=Math.max(...values.map(m=>m.end));
 const blocks=values.flatMap(m=>m.cds_blocks.map(b=>[b[0],b[1]])).sort((a,b)=>a[0]-b[0]);
 const union=[];for(const b of blocks){const last=union.at(-1);if(last&&b[0]<=last[1])last[1]=Math.max(last[1],b[1]);else union.push([...b])}
 let cursor=lo;const segments=[];
 for(const [a,b]of union){if(a>cursor)segments.push({a:cursor,b:a,gap:a-cursor>200});segments.push({a,b,gap:false});cursor=b}
 if(cursor<hi)segments.push({a:cursor,b:hi,gap:hi-cursor>200});
 const compressed=mode==='exons',breaks=compressed?segments.filter(s=>s.gap).length:0;
 const width=Math.max(120,availableWidth),gapWidth=breaks?Math.min(22,(width-40)*.35/breaks):0;
 const native=compressed?segments.filter(s=>!s.gap).reduce((n,s)=>n+s.b-s.a,0):hi-lo;
 const px=(width-40-breaks*gapWidth)/Math.max(1,native);let x=20;
 for(const s of segments){s.x=x;s.w=compressed&&s.gap?gapWidth:(s.b-s.a)*px;x+=s.w}
 const map=v=>{if(!compressed)return 20+(v-lo)*px;const s=segments.find(s=>v>=s.a&&v<=s.b)||segments.at(-1);return s.x+(v-s.a)/(s.b-s.a)*s.w};
 return {lo,hi,width,px,segments,map,compressed};
}
// Model-specific CDS exon intervals and intervening genomic introns.
// Coordinates stay 0-based half-open internally; UI tables use 1-based inclusive.
function cdsModelSegments(m){
 const exons=[];
 for(const b of [...m.cds_blocks].sort((a,b)=>a[0]-b[0]||a[1]-b[1])){const last=exons.at(-1);if(last&&b[0]<=last.end)last.end=Math.max(last.end,b[1]);else exons.push({start:b[0],end:b[1]})}
 const reverse=m.strand==='-',ordered=reverse?[...exons].reverse():exons,segments=[];
 ordered.forEach((e,i)=>{segments.push({...e,type:'exon',label:'E'+(i+1),length:e.end-e.start});if(i+1<ordered.length){const next=ordered[i+1],start=Math.min(e.end,next.end),end=Math.max(e.start,next.start);if(end>start)segments.push({start,end,type:'intron',label:'I'+(i+1),length:end-start})}});
 const cds=exons.reduce((n,e)=>n+e.end-e.start,0),span=exons.length?exons.at(-1).end-exons[0].start:0;
 return {segments,cds,span,introns:span-cds,start:exons[0]?.start,end:exons.at(-1)?.end,exonCount:exons.length};
}
function cdsIntronDisplay(segment,scale){
 if(!scale.compressed)return 'true genomic scale';
 const compressed=scale.segments.filter(s=>s.gap).reduce((n,s)=>n+Math.max(0,Math.min(segment.end,s.b)-Math.max(segment.start,s.a)),0);
 return compressed===0?'true bp scale':compressed===segment.length?'shortened on screen':'partly shortened on screen';
}
