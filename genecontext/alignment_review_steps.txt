function noteSyntenyCheckpoint(pair){

}

  const rows=window.GCFSyntenySelectedRows(h,id);
  const expected=checkpointState(h.alignment_result_id,h.tree_result_id,rows);
  if(id===liveGroup)return !!livePair&&livePair===expected;
  const observed=localStorage.getItem(syntenyCheckKey(id));
  if(observed===expected)return true;
  if(observed!==null&&observed!==h.alignment_result_id+' '+h.tree_result_id)return false;
