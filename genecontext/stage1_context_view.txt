function comparisonRole(l,r){if(!r)return'unavailable';if(l.genome_id!==r.genome_id)return'cross-species';const sameGene=l.annotation_gene_id===r.gene_id,sameRegion=l.contig===r.contig&&l.strand===r.strand&&Math.max(l.start,r.start)<Math.min(l.end,r.end);return sameGene||sameRegion?'self-control':'same-assembly'}

function contextLabel(l){const e=l.early_synteny;if(!e)return'Neighbor context not calculated';if(!e.comparisons?.length)return'Neighbor context unavailable';const own=crossComparisons(l);return own.length?`Cross-species neighbor proposals: ${Math.max(...own.map(c=>c.matches.length))} in best comparison`:'Only within-assembly controls available'}
function choiceKey(l,r){return JSON.stringify([l.genome_id,l.locus_id,r.reference_id])}
function choices(l,r){const key=choiceKey(l,r);if(!neighborChoices.has(key))neighborChoices.set(key,{genome_id:l.genome_id,locus_id:l.locus_id,reference_id:r.reference_id,reference_genes:[],candidate_genes:[],pairs:[]});return neighborChoices.get(key)}
function pairKey(m){return JSON.stringify([m.reference_gene,m.target_gene])}
function shortlist(){return[...neighborChoices.values()].filter(c=>c.reference_genes.length||c.candidate_genes.length||c.pairs.length)}
function loadNeighborChoices(rows){const validated=[];for(const c of rows||[]){const l=lmap.get(c.locus_id),r=refsById.get(c.reference_id);if(!l||l.genome_id!==c.genome_id||!r)throw Error('Unknown locus/reference in neighbor shortlist.');const comp=l.early_synteny?.comparisons.find(x=>x.reference_id===c.reference_id);if(!comp)throw Error('Missing comparison in neighbor shortlist.');const rg=new Set(r.neighbors.map(g=>g.gene_id)),tg=new Set(l.early_synteny.neighbors.map(g=>g.gene_id)),pk=new Set(comp.matches.map(pairKey));for(const[field,allowed]of[['reference_genes',rg],['candidate_genes',tg],['pairs',pk]])if(!Array.isArray(c[field])||new Set(c[field]).size!==c[field].length||c[field].some(id=>!allowed.has(id)))throw Error('Unknown/duplicate neighbor choice.');validated.push({...c,reference_genes:[...c.reference_genes],candidate_genes:[...c.candidate_genes],pairs:[...c.pairs]})}neighborChoices.clear();for(const c of validated)neighborChoices.set(JSON.stringify([c.genome_id,c.locus_id,c.reference_id]),c)}

function geneRange(anchor,g){const mid=(anchor.start+anchor.end)/2,sign=anchor.strand==='-'?-1:1;return[(g.start-mid)*sign,(g.end-mid)*sign].sort((a,b)=>a-b)}
function relativeDistance(anchor,g){if(g.target_side==='upstream')return'-'+((g.distance_bp??Math.max(0,anchor.strand==='+'?anchor.start-g.end:g.start-anchor.end))/1000).toFixed(1)+' kb';return'+'+((g.distance_bp??Math.max(0,anchor.strand==='+'?g.start-anchor.end:anchor.start-g.end))/1000).toFixed(1)+' kb'}

function synteny(l){
 const box=el('div'),value=l.early_synteny;
 if(!value){box.append(el('p','Neighbor context has not been calculated for this checkpoint.','status'));return box}
 if(!value.comparisons?.length){box.append(el('p','No reference comparison is available.','status'),el('p','Saved reason: '+value.status.replaceAll('_',' ')+'. Missing context does not establish absence of the target or its neighbors.','legend'));return box}
