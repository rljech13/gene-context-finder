function referenceAvailability(l){
 const refs=DATA.early_synteny?.references||[],resolution=new Map((DATA.early_synteny?.source_resolution||[]).map(r=>[r.query,r])),rows=[];
 for(const q of DATA.queries)for(const m of q.members){const resolved=refs.filter(r=>(r.queries||[]).includes(m.query)),record=resolution.get(m.query);let reason;
  if(resolved.length){reason=resolved.map(r=>`${species(r.genome_id)} · ${r.name||r.gene_id}`).join('; ');if(record?.context_binding_only)reason+=' · linked by curated GeneID; query sequence differs from this annotation release.'}
  else if(record?.status==='identifier_sequence_mismatch')reason='Sequence differs from the supplied annotation; this saved checkpoint did not use a curated GeneID to bind its source locus.';
  else if(record?.status==='coding_reference_unavailable')reason='A coding source reference could not be established for this query.';
  else if(record?.sources?.length)reason='Source locus resolved, but its neighborhood could not be prepared.';
  else reason='No source locus resolved in the supplied source datasets.';
  rows.push([m.query,m.species||'Not supplied',resolved.length?'Available':'Unavailable',reason]);
 }
 return table(['Query','Source species','Reference neighborhood','Reason / shared source locus'],rows);
}

 const comparisons=l.early_synteny.comparisons||[],rows=[{anchor:l,role:'Candidate',neighbors:[]}];
 for(const c of comparisons){const r=refsById.get(c.reference_id);if(r)rows.push({anchor:r,role:comparisonRole(l,r)==='cross-species'?'Reference':comparisonRole(l,r)==='self-control'?'Self-control':'Same assembly',comparison:c,neighbors:[]})}
 rows[0].neighbors=l.early_synteny.neighbors||[];
 for(const row of rows.slice(1))row.neighbors=(row.anchor.neighbors||[]).filter(g=>g.side_rank<=l.early_synteny.neighbors_used);
 for(const row of rows)row.neighbors=row.neighbors.filter(g=>scope==='window'?g.end>row.anchor.start-DATA.config.window&&g.start<row.anchor.end+DATA.config.window:g.side_rank<=Number(scope));
 const candidateIds=new Set(rows[0].neighbors.map(g=>g.gene_id)),matchedIds=new Set();
 for(const row of rows.slice(1)){const visible=new Set(row.neighbors.map(g=>g.gene_id));row.visibleMatches=row.role==='Reference'?row.comparison.matches.filter(m=>candidateIds.has(m.target_gene)&&visible.has(m.reference_gene)):[];for(const m of row.visibleMatches)matchedIds.add(m.target_gene)}
 // A color is a candidate-gene key for this diagram, not an asserted orthogroup.
 const candidateOrder=[...rows[0].neighbors].sort((a,b)=>geneRange(l,a)[0]-geneRange(l,b)[0]),colors=new Map(candidateOrder.filter(g=>matchedIds.has(g.gene_id)).map((g,i)=>[g.gene_id,matchedIds.size<=PAIR_COLORS.length?PAIR_COLORS[i]:`hsl(${(i*137.508)%360} 52% 38%)`]));
 const dense=rows.some(row=>row.neighbors.length>10),rowHeight=dense?150:108,width=1260,center=630,left=100,right=1160,height=rowHeight*rows.length+30;
 const svg=svgEl('svg',{viewBox:`0 0 ${width} ${height}`,class:'neighbor-map',role:'img','aria-label':'Candidate neighborhood compared with all available query-source loci'});
 const maxSlots=Math.max(1,...rows.flatMap(row=>['upstream','downstream'].map(side=>row.neighbors.filter(g=>g.target_side===side).length))),step=Math.min(88,(right-center-24)/maxSlots),extent=Math.max(1,...rows.map(row=>(row.anchor.end-row.anchor.start)/2+DATA.config.window),...(scope==='window'?[]:rows.flatMap(row=>row.neighbors.flatMap(g=>geneRange(row.anchor,g).map(Math.abs))))),bx=v=>center+v/extent*(right-center-15);
 svg.append(svgEl('line',{x1:center,x2:center,y1:27,y2:height-30,stroke:'#d6c8e3','stroke-dasharray':'4 4'}));
 rows.forEach((row,index)=>{const a=row.anchor,offset=index*rowHeight,y=offset+(dense?115:78),genes=[...row.neighbors].sort((g,h)=>geneRange(a,g)[0]-geneRange(a,h)[0]),up=genes.filter(g=>g.target_side==='upstream'),down=genes.filter(g=>g.target_side==='downstream'),byRef=new Map((row.visibleMatches||[]).map(m=>[m.reference_gene,m]));
  const heading=svgEl('text',{x:20,y:offset+18,'font-size':13,'font-weight':index?500:700},`${row.role}: ${species(a.genome_id)} · ${a.annotation_gene_name||a.name||a.gene_id||DATA.config.gene}`);heading.append(svgEl('title',{},`${a.contig}:${a.start}–${a.end} (${a.strand})\n${(a.queries||[]).join(', ')}`));svg.append(heading);
  if(index){const c=row.comparison,label=row.role==='Reference'?`${row.visibleMatches.length} shown / ${c.matches.length} proposed pairs`:'Control only · not cross-species evidence';svg.append(svgEl('text',{x:right,y:offset+18,'font-size':11,'text-anchor':'end',fill:'#5a6b7b'},label))}
  svg.append(svgEl('line',{x1:left,x2:right,y1:y,y2:y,stroke:'#bcc8d4'}));
  const clip=g=>{const r=geneRange(a,g),lim=(a.end-a.start)/2+DATA.config.window;return scope==='window'?[Math.max(-lim,r[0]),Math.min(lim,r[1])]:r};
  const items=[...up.map((g,i)=>({g,slot:i-up.length})),{g:{...a,isTarget:true},slot:0},...down.map((g,i)=>({g,slot:i+1}))],lanes=Array(dense?3:2).fill(-Infinity);
  for(const {g,slot}of items){const range=clip(g),mid=(range[0]+range[1])/2,x=mode==='order'?center+slot*step:bx(mid),labelX=center+slot*step,match=byRef.get(g.gene_id),color=g.isTarget?'#754f9b':index?(match?colors.get(match.target_gene):'#aab8c6'):(colors.get(g.gene_id)||'#aab8c6'),w=Math.min(32,step*.62),x0=mode==='distance'?bx(range[0]):x-w/2,x1=mode==='distance'?Math.max(x0+5,bx(range[1])):x+w/2,tip=Math.min(7,(x1-x0)*.4),forward=g.strand===a.strand,points=forward?`${x0},${y-6} ${x1-tip},${y-6} ${x1},${y} ${x1-tip},${y+6} ${x0},${y+6}`:`${x1},${y-6} ${x0+tip},${y-6} ${x0},${y} ${x0+tip},${y+6} ${x1},${y+6}`;
   const arrow=svgEl('polygon',{points,fill:color,stroke:'white','stroke-width':1});arrow.append(svgEl('title',{},`${g.isTarget?'Target':g.name||g.gene_id}\n${g.contig}:${g.start}–${g.end} (${g.strand})${g.isTarget?'':'\n'+relativeDistance(a,g)}${match?'\nProposed pair: '+match.reference_name+' → '+match.target_name+'; orthology unverified':''}`));svg.append(arrow);
   const full=g.isTarget?'TARGET':g.name||g.gene_id,max=dense?10:15,label=full.length>max?full.slice(0,max-1)+'…':full,span=label.length*6.3+10;let lane=lanes.findIndex(end=>end+5<labelX-span/2);if(lane<0)lane=Math.abs(slot)%lanes.length;lanes[lane]=labelX+span/2;const ly=y-23-lane*20;
   if(mode==='distance')svg.append(svgEl('line',{x1:x,y1:y-9,x2:labelX,y2:ly+4,stroke:'#c1cbd6','stroke-width':1}));
   const text=svgEl('text',{x:labelX,y:ly,'font-size':11,'text-anchor':'middle',fill:g.isTarget?'#633d88':'#203044'},label);text.append(svgEl('title',{},full));svg.append(text);
  }
 });
 if(mode==='distance')for(const f of[-1,-.5,0,.5,1])svg.append(svgEl('text',{x:bx(extent*f),y:height-7,'font-size':10,'text-anchor':'middle',fill:'#5a6b7b'},(extent*f/1000).toFixed(0)+' kb'));
 else svg.append(svgEl('text',{x:center,y:height-7,'font-size':11,'text-anchor':'middle',fill:'#5a6b7b'},'Gene order · equal spacing · each TARGET points right'));
 return svg;
}
function synteny(l){
 const box=el('div'),value=l.early_synteny;if(!value){box.append(el('p','Neighbor context has not been calculated for this checkpoint.','context-note'));return box}
 if(value.status==='protein_matching_unavailable')box.append(el('p','Neighbor protein matching was unavailable; empty match rows are incomplete evidence.','status'));
 const comparisons=value.comparisons||[],refs=comparisons.filter(c=>refsById.has(c.reference_id)),total=DATA.queries.reduce((n,q)=>n+q.members.length,0),resolved=new Set((DATA.early_synteny?.references||[]).flatMap(r=>r.queries||[])),missing=DATA.queries.flatMap(q=>q.members).filter(m=>!resolved.has(m.query)).length;
 box.append(el('p',`${refs.length} available reference neighborhoods · ${missing} / ${total} query variants without a source neighborhood`,'context-note'));
 const diagram=el('div',undefined,'scroll');if(refs.length)box.append(el('p','Matching colors = proposed neighbor pairs to this candidate. Purple = target. Orthology unverified.','context-note'),diagram);else box.append(el('p','No reference comparison available: '+value.status.replaceAll('_',' ')+'.','status'));
 const advanced=el('details');advanced.append(el('summary','How this is checked, missing references and pair details'));
 const cfg=DATA.early_synteny?.parameters||{};advanced.append(el('p',`For each resolved query-source locus, take its annotated neighboring genes. Search their proteins against the candidate species' full protein catalog, then check which matches occur beside this candidate and compare their order, side and direction. Search extent: initially ${cfg.neighbors??5} genes per side, up to ${cfg.max_neighbors??20} within ${(cfg.max_distance??1000000)/1000} kb. This is separate from the later ±${DATA.config.window/1000} kb target window.`,'legend'),el('p',`Preliminary protein gates: ≥${percent(cfg.min_coverage??.5)} coverage of both proteins, ≥${percent(cfg.min_identity??.3)} identity, E ≤ ${cfg.evalue??1e-5}. Competing near-best and many-to-one assignments do not count as proposed pairs. Symbols alone are not used to match genes. Reference variants from the same annotated gene share one neighborhood.`,'legend'),referenceAvailability(l));
 if(refs.length){
  const controls=el('div',undefined,'controls'),scopeLabel=el('label','Overview genes shown'),scope=el('select');scope.setAttribute('aria-label','Overview genes shown');scope.append(Object.assign(el('option',`Within ±${DATA.config.window/1000} kb of each target`),{value:'window'}));for(const n of[5,10,20].filter(n=>n<=value.neighbors_used))scope.append(Object.assign(el('option',`Up to ${n} saved neighbors per side`),{value:String(n)}));if(value.neighbors_used&&!['5','10','20'].includes(String(value.neighbors_used)))scope.append(Object.assign(el('option',`Up to ${value.neighbors_used} saved neighbors per side`),{value:String(value.neighbors_used)}));scope.value=String(Math.min(5,value.neighbors_used));if(![...scope.options].some(o=>o.value===scope.value))scope.value='window';scopeLabel.append(scope);
  const modeLabel=el('label','Overview scale'),mode=el('select');mode.setAttribute('aria-label','Overview scale');mode.append(Object.assign(el('option','Gene order'),{value:'order'}),Object.assign(el('option','Genomic distance (kb)'),{value:'distance'}));modeLabel.append(mode);controls.append(scopeLabel,modeLabel);advanced.append(controls);
  const draw=()=>diagram.replaceChildren(contextOverviewDiagram(l,scope.value,mode.value));scope.addEventListener('change',draw);mode.addEventListener('change',draw);draw();
  const pairs=el('details');pairs.append(el('summary','Inspect one reference: protein matches and neighbor choices'),pairSynteny(l));advanced.append(pairs);
 }
 box.append(advanced);return box;
}
