function term(label,key){const a=el('abbr',label+' ⓘ','metric-term');a.title=METRICS[key]?.[1]||key;a.tabIndex=0;return a}

const refsById=new Map((DATA.early_synteny?.references||[]).map(r=>[r.reference_id,r]));
const neighborChoices=new Map();
function comparisonRole(l,r){if(!r)return'unavailable';if(l.genome_id!==r.genome_id)return'cross-species';const sameGene=l.annotation_gene_id===r.gene_id,sameRegion=l.contig===r.contig&&l.strand===r.strand&&Math.max(l.start,r.start)<Math.min(l.end,r.end);return sameGene||sameRegion?'self-control':'same-assembly'}


function choiceKey(l,r){return JSON.stringify([l.genome_id,l.locus_id,r.reference_id])}
function choices(l,r){const key=choiceKey(l,r);if(!neighborChoices.has(key))neighborChoices.set(key,{genome_id:l.genome_id,locus_id:l.locus_id,reference_id:r.reference_id,reference_genes:[],candidate_genes:[],pairs:[]});return neighborChoices.get(key)}
function pairKey(m){return JSON.stringify([m.reference_gene,m.target_gene])}
function shortlist(){return[...neighborChoices.values()].filter(c=>c.reference_genes.length||c.candidate_genes.length||c.pairs.length)}
function loadNeighborChoices(rows){const validated=[];for(const c of rows||[]){const l=lmap.get(c.locus_id),r=refsById.get(c.reference_id);if(!l||l.genome_id!==c.genome_id||!r)throw Error('Unknown locus/reference in neighbor shortlist.');const comp=l.early_synteny?.comparisons.find(x=>x.reference_id===c.reference_id);if(!comp)throw Error('Missing comparison in neighbor shortlist.');const rg=new Set([...r.neighbors,...displayGenes(r,{...r,neighbors_used:10},'10')].map(g=>g.gene_id)),tg=new Set([...l.early_synteny.neighbors,...displayGenes(l,l.early_synteny,'10')].map(g=>g.gene_id)),pk=new Set(comp.matches.map(pairKey));for(const[field,allowed]of[['reference_genes',rg],['candidate_genes',tg],['pairs',pk]])if(!Array.isArray(c[field])||new Set(c[field]).size!==c[field].length||c[field].some(id=>!allowed.has(id)))throw Error('Unknown/duplicate neighbor choice.');validated.push({...c,reference_genes:[...c.reference_genes],candidate_genes:[...c.candidate_genes],pairs:[...c.pairs]})}neighborChoices.clear();for(const c of validated)neighborChoices.set(JSON.stringify([c.genome_id,c.locus_id,c.reference_id]),c)}
function saveNeighborChoice(){persist()}
function geneRange(anchor,g){const mid=(anchor.start+anchor.end)/2,sign=anchor.strand==='-'?-1:1;return[(g.start-mid)*sign,(g.end-mid)*sign].sort((a,b)=>a-b)}
function relativeDistance(anchor,g){if(g.target_side==='upstream')return'-'+((g.distance_bp??Math.max(0,anchor.strand==='+'?anchor.start-g.end:g.start-anchor.end))/1000).toFixed(1)+' kb';return'+'+((g.distance_bp??Math.max(0,anchor.strand==='+'?g.start-anchor.end:anchor.start-g.end))/1000).toFixed(1)+' kb'}

const DISPLAY_NEIGHBOR_LIMIT=10;
function displayGenes(anchor,context,scope){
 const cap=Math.min(DISPLAY_NEIGHBOR_LIMIT,context?.neighbors_used??DISPLAY_NEIGHBOR_LIMIT),n=scope==='window'?cap:Math.min(cap,Number(scope));
 const ordinary=(context?.neighbors||anchor.neighbors||[]).filter(g=>g.side_rank<=n);
 const withinWindow=g=>g.end>anchor.start-DATA.config.window&&g.start<anchor.end+DATA.config.window;
 const shown=scope==='window'?ordinary.filter(withinWindow):ordinary;
 const lo=Math.min(anchor.start-DATA.config.window,...shown.map(g=>g.start)),hi=Math.max(anchor.end+DATA.config.window,...shown.map(g=>g.end));
 const extras=[...(context?.display_only_genes||anchor.display_only_genes||[])];
 // Historical checkpoints did not retain omitted source copies. Their exact
 // annotation coordinates are available in the saved reference list.
 for(const r of refsById.values())if(r.genome_id===anchor.genome_id&&r.contig===anchor.contig&&r.gene_id!==(anchor.annotation_gene_id||anchor.gene_id)){
  if(!extras.some(g=>g.gene_id===r.gene_id))extras.push({...r,display_only:true,display_reason:'other_query_source'});
 }
 const known=new Set(shown.map(g=>g.gene_id));
 for(const g of extras){if(known.has(g.gene_id)||!(scope==='window'?withinWindow(g):g.end>lo&&g.start<hi))continue;
  const relative=((g.start+g.end)-(anchor.start+anchor.end))*(anchor.strand==='-'?-1:1);
  shown.push({...g,display_only:true,target_side:relative<0?'upstream':'downstream',side_rank:0});known.add(g.gene_id);
 }
 return shown;
}
function plotGenes(l,r,scope){return[{anchor:r,role:'Reference',neighbors:displayGenes(r,{...r,neighbors_used:l.early_synteny.neighbors_used},scope)},{anchor:l,role:'Candidate',neighbors:displayGenes(l,l.early_synteny,scope)}]}

function visibleMatches(rows,c){const a=new Set(rows[0].neighbors.map(g=>g.gene_id)),b=new Set(rows[1].neighbors.map(g=>g.gene_id));return c.matches.filter(m=>a.has(m.reference_gene)&&b.has(m.target_gene))}
function uncertainEdges(l,c){
 const byProtein=new Map((l.early_synteny?.neighbors||[]).filter(g=>g.protein_id).map(g=>[g.protein_id,g]));
 const rows=(c.colliding_matches||[]).map(m=>({...m,reason:'Several reference neighbors map to this candidate gene; not independent anchors.'}));
 for(const a of c.ambiguous_matches||[])for(const m of a.matches||[]){const g=byProtein.get(m.target_protein);if(g)rows.push({...m,reference_gene:a.reference_gene,target_gene:g.gene_id,target_name:g.name,reason:'Several near-best protein hits compete for this assignment.'})}
 return rows;
}

   if(a.target_gene!==b.target_gene)continue;
   const key=JSON.stringify([a.reference_gene,b.reference_gene,a.target_gene]),p=positions[i-1].get(a.reference_gene),q=positions[i].get(b.reference_gene);if(seen.has(key)||!p||!q)continue;seen.add(key);
   const kind=a.kind==='unresolved'||b.kind==='unresolved'?'unresolved':a.kind==='control'||b.kind==='control'?'control':'retained',color=kind==='unresolved'?'#bd7709':colors.get(a.target_gene)||'#52708c',delta=q.y-p.y,path=svgEl('path',{d:`M ${p.x} ${p.y+8} C ${p.x} ${p.y+delta*.4}, ${q.x} ${p.y+delta*.6}, ${q.x} ${q.y-8}`,fill:'none',stroke:color,'stroke-width':kind==='unresolved'?2:3,'stroke-dasharray':kind==='unresolved'?'3 4':kind==='control'?'7 4':'none',class:'neighbor-link','data-link-kind':kind,'data-candidate-gene':a.target_gene,'data-from-row':i-1,'data-to-row':i,'data-reference-gene':b.reference_gene});
   path.append(svgEl('title',{},`${p.name} ↔ ${q.name}\nShared candidate neighbor: ${positions[0].get(a.target_gene)?.name||a.target_gene}\n${kind==='unresolved'?'Unresolved assignment; excluded from independent anchors.':kind==='control'?'Includes a same-assembly control; not independent cross-species evidence.':'Saved proposed protein correspondences; orthology unverified.'}${i>1?'\nReference rows share a candidate correspondence; they were not searched directly against each other.':''}`));linkLayer.append(path);register(path,[a.target_gene]);
  }
 }
 if(mode==='distance')for(const f of[-1,-.5,0,.5,1])svg.append(svgEl('text',{x:bx(extent*f),y:height-7,'font-size':10,'text-anchor':'middle',fill:'#5a6b7b'},(extent*f/1000).toFixed(0)+' kb'));
 else svg.append(svgEl('text',{x:center,y:height-7,'font-size':11,'text-anchor':'middle',fill:'#5a6b7b'},'Gene order · equal spacing · each TARGET points right'));
 return svg;
}
function synteny(l){
 const box=el('div'),value=l.early_synteny;
 if(!value)return el('p','Neighbor context has not been calculated.','context-note');

 if(!refs.length){const reason={annotation_unavailable:'The candidate assembly has no supplied annotation, so its neighboring genes cannot be compared at this stage.',dataset_error:'The candidate annotation could not be prepared; inspect the dataset error.',annotation_contig_missing:'The candidate contig is absent from the supplied annotation.',annotation_contig_length_mismatch:'Candidate and annotation contig lengths differ.',source_locus_unresolved:'No query-source neighborhood was resolved. Inspect Query origins and supplied datasets.',protein_catalog_empty:'The candidate annotation has no usable protein catalogue.'}[value.status]||'No source-neighborhood comparison is available: '+value.status.replaceAll('_',' ')+'.';return el('p',reason+' This does not establish gene absence.','context-note')}
 const resolvedCount=(DATA.early_synteny.source_resolution||[]).filter(r=>r.sources?.length).length,totalCount=DATA.queries.reduce((n,q)=>n+q.members.length,0),crossCount=refs.filter(c=>comparisonRole(l,refsById.get(c.reference_id))==='cross-species').length;

 if(value.status==='protein_matching_unavailable')box.append(el('p','Protein matching was unavailable; empty links are incomplete evidence.','status'));
 const controls=el('div',undefined,'controls'),scopeLabel=el('label','Genes shown'),scope=el('select');scope.setAttribute('aria-label','Overview genes shown');
 scope.append(Object.assign(el('option',`Within ±${DATA.config.window/1000} kb`),{value:'window'}));

Prediction window later: ±${DATA.config.window/1000} kb.${value.neighbors_used>DISPLAY_NEIGHBOR_LIMIT?'  box.append(el('p','Lines: proposed protein matches. Dashed: same-assembly controls. Amber: unresolved. Purple: target; purple outline: other target copy or overlapping annotation, excluded from anchors. Gray genes remain visible. Hover to trace a match.','context-note'));
 box.append(lazyDetails('How neighbors are matched',()=>{const cfg=DATA.early_synteny.parameters,details=el('div');details.append(el('p',`Proteins of annotated neighbors at every available query-source locus are searched against the candidate species’ full protein catalog. Local matches are then compared by order, target-relative side and direction. Competing near-best and many-to-one assignments remain unresolved. Source availability and sequence mismatches are listed once under Query origins.`,'legend'));return details}),lazyDetails('Protein match details and optional neighbor follow-up',()=>pairSynteny(l)));
 return box;
}
