function validateFinal(s){
 if(!s||s.schema!=='stage3_final_choices_v1'||s.review_id!==finalSpec.review_id||s.set_id!==finalSpec.set_id)throw Error('These choices belong to a different reviewed sequence set.');
 if(!s.choices||Object.keys(s.choices).sort().join('|')!==finalSpec.buckets.map(b=>b.key).sort().join('|'))throw Error('The complete gene / organism set is required.');
 for(const b of finalSpec.buckets){const c=s.choices[b.key];if(!c||!['',...b.model_ids].includes(c.primary)||typeof c.reason!=='string'||!c.reason.trim()||typeof c.reviewed!=='boolean')throw Error('Each organism needs a valid primary model (or none), a reason and a review status.');}
 const ids=finalSpec.buckets.flatMap(b=>b.model_ids);if(s.cases!==undefined&&(!s.cases||Array.isArray(s.cases)||typeof s.cases!=='object'))throw Error('Invalid separate-case notes.');
 for(const[id,c]of Object.entries(s.cases||{}))if(!ids.includes(id)||!c||typeof c.reason!=='string'||!c.reason.trim()||typeof c.reviewed!=='boolean')throw Error('Invalid separate-case note.');
 return JSON.parse(JSON.stringify(s));
}
try{const saved=localStorage.getItem(finalKey);if(saved)finalState=validateFinal(JSON.parse(saved))}catch(e){$('finalMessage').textContent='Saved choices were not loaded: '+e.message}
function persistFinal(){try{localStorage.setItem(finalKey,JSON.stringify(finalState));$('finalMessage').textContent='Saved in this browser. Download the split package to share these choices.'}catch(e){$('finalMessage').textContent='Browser storage is unavailable. Download the split package to keep these choices.'}}
function automaticCaseReason(m,primary){const a=m.assignment;if(a.known_other_copy)return 'Specific source label is '+m.copy_identity.display_name+'; it is not automatically renamed to '+m.target_gene+'.';if(a.alternate_preferred){const r=a.best_alternate;return 'Reference preference favors '+r.symbol+' ('+r.accession+'); kept separately as a possible related copy, not assigned cross-species orthology.'}return primary?'Additional reviewed copy; '+modelMap.get(primary).original_model+' is the current target representative in this organism.':'No target representative assigned in this organism; copy identity remains a separate case.'}
function finalRows(){const primary=[],cases=[];for(const b of finalSpec.buckets){const c=finalState.choices[b.key];for(const id of b.model_ids){const m=modelMap.get(id),yes=c.primary===id,note=yes?c:finalState.cases?.[id],r=m.assignment.best_target||{},identity=m.copy_identity;const row={gene:b.gene,group:b.group,species:m.species,genome_id:m.genome_id,assembly_accession:m.assembly_accession||'',model_id:id,original_model:m.original_model,original_gene:m.original_gene||'',copy_label:m.copy_label||'',copy_identity:identity.display_name,copy_identity_status:identity.status,copy_identity_basis:identity.basis,ncbi_gene_id:identity.ncbi_gene_id,source:m.source,contig:m.contig,start:m.start,end:m.end,strand:m.strand,protein_aa:m.protein_aa,classification:yes?'primary_target':'special_case',selection_reason:note?note.reason:automaticCaseReason(m,c.primary),decision_origin:note?.reviewed?'reviewed':'suggestion',best_target_reference:r.accession||'',reference_gene:r.symbol||'',protein_identity:r.identity??'',model_coverage:r.coverage_model??'',reference_coverage:r.coverage_reference??'',evidence:m.assignment.facts.join(' '),cds_file:m.cds_file||'',protein_file:m.protein_file||''};(yes?primary:cases).push(row)}}return{primary,cases}}
function finalTable(rows,isPrimary){const table=el('table',undefined,'decision-table'),head=el('thead'),h=el('tr');for(const s of ['Organism / copy','Model & locus',isPrimary?'Basis for target assignment':'Why kept separately','Sequences / choice'])h.append(el('th',s));head.append(h);table.append(head);const body=el('tbody');for(const r of rows){const m=modelMap.get(r.model_id),tr=el('tr');tr.dataset.retainedModelId=r.model_id;const org=el('td');org.append(el('i',r.species),el('span',r.copy_identity,'identity-label'),el('small',r.decision_origin==='reviewed'?'Reviewed assignment':'Proposed assignment','assignment-status'));const model=el('td');model.append(el('strong',r.original_model,'model-accession'),el('small',(r.source==='tiberius'?'Tiberius':'Annotation')+' · '+r.protein_aa+' aa','assignment-status'),el('span',r.contig+':'+(Number(r.start)+1).toLocaleString('en-US')+'–'+Number(r.end).toLocaleString('en-US')+' ('+r.strand+')','accession'));if(r.assembly_accession)model.append(el('span',r.assembly_accession,'accession'));const basis=el('td');basis.append(el('p',r.selection_reason,'assignment-reason'));const detail=el('details'),summary=el('summary','Evidence');detail.append(summary);for(const fact of m.assignment.facts)detail.append(el('p',fact));if(m.assignment.best_alternate){const a=m.assignment.best_alternate;detail.append(el('p','Related-copy reference: '+a.symbol+' · '+a.accession+' · '+pct(a.identity)+' identity; coverage '+pct(a.coverage_model)+' / '+pct(a.coverage_reference)+'.'))}basis.append(detail);const files=el('td'),dl=el('div',undefined,'downloads');for(const[field,label]of [['cds_file','CDS'],['protein_file','Protein']])if(r[field]){const a=link(r[field],label,'download');a.download='';dl.append(a)}files.append(dl);const button=el('button',isPrimary?'Change choice':'Use as target','choice-button');button.type='button';button.addEventListener('click',()=>editPrimary(r.model_id));files.append(button);if(!isPrimary){const edit=el('button','Edit note','choice-button');edit.type='button';edit.addEventListener('click',()=>editCase(r.model_id));files.append(edit)}tr.append(org,model,basis,files);body.append(tr)}table.append(body);return rows.length?table:el('p',isPrimary?'No target representative assigned.':'No separate cases in this selection.','empty')}
function finalChoices(){const all=finalRows(),q=$('decisionSearch').value.trim().toLowerCase(),filter=r=>r.group===group()&&[r.species,r.original_model,r.copy_identity,r.contig].join(' ').toLowerCase().includes(q),p=all.primary.filter(filter),c=all.cases.filter(filter),b=finalSpec.buckets.filter(b=>b.group===group());$('decisionCount').textContent=all.primary.filter(r=>r.group===group()).length+' target representatives · '+all.cases.filter(r=>r.group===group()).length+' separate cases · '+b.filter(b=>finalState.choices[b.key].reviewed).length+' / '+b.length+' organism assignments reviewed';$('decisionTable').replaceChildren(finalTable(p,true));$('specialCaseTable').replaceChildren(finalTable(c,false));$('primaryCount').textContent=p.length;$('caseCount').textContent=c.length;const unresolved=b.filter(b=>!finalState.choices[b.key].primary);$('unassignedTargets').replaceChildren();if(unresolved.length){$('unassignedTargets').append(el('p','No primary assigned:','assignment-status'));for(const x of unresolved){const btn=el('button',x.species);btn.addEventListener('click',()=>editPrimary(x.model_ids[0]));$('unassignedTargets').append(btn)}}}
function editorEvidence(){const m=modelMap.get($('finalModel').value);$('finalEvidence').textContent=m?m.copy_identity.display_name+'. '+m.assignment.facts.join(' '):'All selected copies will remain in Separate cases; no target sequence will be exported for this organism.'}

function finalTSV(rows){const fields=Object.keys((rows[0]||DATA.finalization.rows.primary[0]||DATA.finalization.rows.cases[0])||{gene:'',species:'',model_id:'',selection_reason:''});return fields.join('\t')+'\n'+rows.map(r=>fields.map(k=>String(r[k]??'').replace(/[\t\r\n]/g,' ')).join('\t')).join('\n')+(rows.length?'\n':'')}

function zipFiles(files){const encoder=new TextEncoder(),parts=[],central=[];let offset=0;const u16=(v,p,n)=>v.setUint16(p,n,true),u32=(v,p,n)=>v.setUint32(p,n>>>0,true);for(const[name,text]of Object.entries(files)){const filename=encoder.encode(name),data=encoder.encode(text);let crc=0xffffffff;for(const x of data){crc^=x;for(let i=0;i<8;i++)crc=(crc>>>1)^((crc&1)?0xedb88320:0)}crc=(crc^0xffffffff)>>>0;const header=new Uint8Array(30+filename.length),v=new DataView(header.buffer);u32(v,0,0x04034b50);u16(v,4,20);u16(v,6,0x0800);u32(v,14,crc);u32(v,18,data.length);u32(v,22,data.length);u16(v,26,filename.length);header.set(filename,30);parts.push(header,data);const c=new Uint8Array(46+filename.length),cv=new DataView(c.buffer);u32(cv,0,0x02014b50);u16(cv,4,20);u16(cv,6,20);u16(cv,8,0x0800);u32(cv,16,crc);u32(cv,20,data.length);u32(cv,24,data.length);u16(cv,28,filename.length);u32(cv,42,offset);c.set(filename,46);central.push(c);offset+=header.length+data.length}const size=central.reduce((n,a)=>n+a.length,0),end=new Uint8Array(22),v=new DataView(end.buffer);u32(v,0,0x06054b50);u16(v,8,central.length);u16(v,10,central.length);u32(v,12,size);u32(v,16,offset);return new Blob([...parts,...central,end],{type:'application/zip'})}
