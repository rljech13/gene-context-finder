function diagramPairs(){const source=$('modelSource').value;if(source==='all')return DATA.neighbor_pairs;if(DATA.source_neighbor_pairs)return DATA.source_neighbor_pairs[source]||[];return DATA.neighbor_pairs.filter(p=>models.get(p.model_a).source===source&&models.get(p.model_b).source===source)}
function groupRange(r,g){
 const middle=(r.anchor_start+r.anchor_end)/2,sign=r.strand==='-'?-1:1;
 return [(Math.min(...g.map(m=>m.start))-middle)*sign,(Math.max(...g.map(m=>m.end))-middle)*sign].sort((a,b)=>a-b);
}
function windowRange(r){
 const middle=(r.anchor_start+r.anchor_end)/2,sign=r.strand==='-'?-1:1;
 return [(r.window_start-middle)*sign,(r.window_end-middle)*sign].sort((a,b)=>a-b);
}
function anchorHighlights(r){
 const blocks=diagramModels(r).filter(m=>m.overlaps_anchor).flatMap(m=>m.cds_blocks||[]).map(b=>[Math.max(b[0],r.window_start),Math.min(b[1],r.window_end)]).filter(b=>b[0]<b[1]).sort((a,b)=>a[0]-b[0]),merged=[];
 for(const [a,b]of blocks){if(merged.length&&a<=merged.at(-1)[1])merged.at(-1)[1]=Math.max(merged.at(-1)[1],b);else merged.push([a,b])}
 return merged.length?merged.map(([start,end])=>({start,end,kind:'coding_segments'})):[{start:r.anchor_start,end:r.anchor_end,kind:'search_interval'}];
}
function sharedGenomicRuler(zoom){
 // Fit the common native-coordinate extent into the panel at 1x, as in
 // the NANOG overview. Every row and source uses this same physical ruler.
 const viewerWidth=$('links').clientWidth||$('links').getBoundingClientRect().width||1260;
 const rawExtent=Math.max(50000,...DATA.regions.flatMap(r=>windowRange(r).map(Math.abs)));
 const magnitude=10**Math.floor(Math.log10(rawExtent/5)),normal=rawExtent/5/magnitude;
 const major=(normal<=1?1:normal<=2?2:normal<=5?5:10)*magnitude;
 const extent=Math.ceil(rawExtent/major)*major,left=250;
 const plotWidth=Math.max(160,viewerWidth-300)*zoom,pxPerBp=plotWidth/(2*extent),right=left+plotWidth;
 return {pxPerBp,extent,major,minor:major/5,left,right,center:(left+right)/2,width:right+50};
}

function sourceCDSAgreement(g){

function neighborhood(r){
 const limit=Number($('neighborsShown').value),all=diagramGroups(r),anchor=all.filter(g=>g.some(m=>m.overlaps_anchor));
 const others=all.filter(g=>!g.some(m=>m.overlaps_anchor));
 const left=others.filter(g=>{const [a,b]=groupRange(r,g);return (a+b)/2<0}).sort((a,b)=>groupRange(r,a)[0]-groupRange(r,b)[0]).slice(-limit);
 const right=others.filter(g=>{const [a,b]=groupRange(r,g);return (a+b)/2>=0}).sort((a,b)=>groupRange(r,a)[0]-groupRange(r,b)[0]).slice(0,limit);
 const hiddenLeft=others.filter(g=>{const [a,b]=groupRange(r,g);return (a+b)/2<0}).length-left.length;
 const hiddenRight=others.length-left.length-right.length-hiddenLeft;
 return {anchor:anchor.sort((a,b)=>groupRange(r,a)[0]-groupRange(r,b)[0]),left,right,hidden: hiddenLeft+hiddenRight,hiddenLeft,hiddenRight};
}

function correspondenceEdges(a,b,allSources=false){
 const groups=allSources?groupModels:diagramGroups,ga=new Set(groups(a).map(g=>g[0].gene_group)),gb=new Set(groups(b).map(g=>g[0].gene_group)),unique=new Map();
 // Keep the saved record orientation intact: the exported pair must still
 // identify the original evidence, even when its endpoints are drawn reversed.
 for(const p of allSources?DATA.neighbor_pairs:diagramPairs()){
  let from,to;
  if(ga.has(p.gene_a)&&gb.has(p.gene_b)){from=p.gene_a;to=p.gene_b}
  else if(ga.has(p.gene_b)&&gb.has(p.gene_a)){from=p.gene_b;to=p.gene_a}
  else continue;
  const key=pairKey(p),old=unique.get(key);
  if(!old||p.bitscore>old.evidence.bitscore)unique.set(key,{from,to,key,evidence:p,kind:p.kind==='target_similarity'?'target':p.status==='ambiguous_proposal'?'unresolved':'retained'});
 }
 if(a.genome_id===b.genome_id)for(const id of ga)if(gb.has(id))unique.set('control:'+id,{from:id,to:id,key:'control:'+id,kind:'control'});
 return [...unique.values()];
}

function rulerGeneLabel(group){
 const names=groupLabel(group).split(' / '),symbols=names.filter(n=>!/^LOC\d+$/.test(n)),ids=names.filter(n=>/^LOC\d+$/.test(n));
 const primary=symbols.length?symbols.join(' / '):ids.join(' / ')||'Unassigned gene';
 const secondary=symbols.length?ids.join(' / '):group[0].context_only?group[0].biotype:'';
 const wrap=text=>{const parts=[];while(text.length>32){let cut=text.lastIndexOf(' ',32);if(cut<12)cut=32;parts.push(text.slice(0,cut));text=text.slice(cut).trim()}if(text)parts.push(text);return parts};
 return [...wrap(primary).map(text=>({text,secondary:false})),...wrap(secondary).map(text=>({text,secondary:true}))];
}
function rulerModelStructures(group){
 const structures=new Map();
 for(const m of group){
  // Equality is checked in full native coordinates before window clipping.
  // Sharing a symbol, overlap or CDS sequence alone does not merge structures.
  const blocks=values=>[...values].map(b=>[...b]).sort((a,b)=>a[0]-b[0]||a[1]-b[1]);
  const key=JSON.stringify([m.contig,m.strand,m.start,m.end,!!m.context_only,m.biotype||'',blocks(m.cds_blocks||[]),blocks(m.stop_blocks||[]),m.cds_sha256||'']);
  if(!structures.has(key))structures.set(key,[]);structures.get(key).push(m);
 }
 return [...structures.values()].sort((a,b)=>a[0].start-b[0].start||a[0].end-b[0].end||a[0].model_id.localeCompare(b[0].model_id));
}
